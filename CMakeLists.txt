cmake_minimum_required(VERSION 3.24)
project(Dlss5Loader VERSION 0.3.3 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with Windows x64 (Visual Studio -A x64).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(loader_core STATIC src/loader.cpp)
target_include_directories(loader_core PUBLIC src)
target_link_libraries(loader_core PUBLIC bcrypt version wintrust crypt32)
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
add_library(imgui STATIC
  ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp
  ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp
  ${IMGUI_DIR}/misc/cpp/imgui_stdlib.cpp
  ${IMGUI_DIR}/backends/imgui_impl_win32.cpp
  ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${IMGUI_DIR} ${IMGUI_DIR}/backends ${IMGUI_DIR}/misc/cpp)
target_link_libraries(imgui PUBLIC d3d11 dxgi dwmapi)
add_executable(Dlss5Loader WIN32 src/main.cpp)
set_target_properties(Dlss5Loader PROPERTIES OUTPUT_NAME "Dlss5Setup")
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
file(TO_CMAKE_PATH "${Python3_EXECUTABLE}" DLSS_PYTHON_PATH)
target_include_directories(Dlss5Loader PRIVATE third_party/json)
target_compile_definitions(Dlss5Loader PRIVATE DLSS_PYTHON_EXECUTABLE="${DLSS_PYTHON_PATH}")
target_link_libraries(Dlss5Loader PRIVATE loader_core imgui comdlg32 shell32)
if(MSVC)
  target_compile_options(loader_core PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(Dlss5Loader PRIVATE /W4 /permissive- /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(LoadTestHost tests/host.cpp)
  add_library(LoadTestDll SHARED tests/dll.cpp)
  add_library(RejectTestDll SHARED tests/dll.cpp)
  target_compile_definitions(RejectTestDll PRIVATE REJECT_ATTACH)
  add_executable(LoaderTests tests/tests.cpp)
  target_link_libraries(LoaderTests PRIVATE loader_core)
  add_test(NAME loadlibrary_integration COMMAND LoaderTests)
  add_test(NAME installer_lifecycle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_backend.py)
  set_tests_properties(installer_lifecycle PROPERTIES LABELS "payload")
  add_test(NAME steam_launch_routing COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_launch.py)
endif()
